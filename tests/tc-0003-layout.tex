\def\TCID{%
  TC-0003
}

\def\TCTitle{%
  Layout of tabs and toolbar buttons
}

\def\TCPurpose{%
  Verify the layout rules of the tabs of a TabStrip and the buttons of a
  Toolbar: default sizes, fixed widths, hidden items, wrapping and the
  grouping of buttons.
}

\def\TCPriority{%
  \TCPriorityHigh
}

\def\TCRequirementRef{%
  REQ-0011, REQ-0018, REQ-0020
}

\def\TCDesignRef{%
  Not applicable; the rules are in the named header.
}

\def\TCFeatureRef{%
  \texttt{include/wfc/mscomctl/layout.hpp} (\texttt{tab\_layout},
  \texttt{button\_layout}, \texttt{button\_group}) and
  \texttt{tests/layout\_test.cpp} and the runner
  \texttt{tests/tc-0003-layout.ps1}
}

\def\TCTestTechnique{%
  Equivalence partitioning and boundary value analysis of the rule functions
  (ISO/IEC/IEEE 29119-4 specification-based techniques).
}

\def\TCPreconditions{%
  The library is built standalone (\texttt{cmake -S . -B out/build}, then
  \texttt{cmake --build out/build --config Debug}) with
  \texttt{WFC\_BUILD\_TESTS} on; CTest runs the test as
  \texttt{TC-0003-layout}
  (or \texttt{wfc-mscomctl-TC-0003-layout} when WFC adds
  the library).
}

\def\TCEnvironment{%
  Any target the library builds for (Windows x86, x64 and ARM64 with MSVC,
  Linux through Winelib); local runs use the x64 Debug configuration of the
  standalone build. The test needs no window, registry or network.
}

\def\TCAssumptions{%
  The rule header under test is the one built into the library; the test is
  compiled from the same source revision.
}

\def\TCInputData{%
  Four tabs (caption lengths 6, 10, hidden, 0), five buttons (default,
  separator, placeholder of width 50, default of width 700, hidden), and style
  lists for the group rule; widths in twips.
}

\def\TCInitialState{%
  A fresh process; the test keeps no state between runs and holds no files or
  handles.
}

\def\TCProcedure{%
  \begin{enumerate}
    \item Run \texttt{mscomctl\_layout\_test} (CTest \texttt{TC-0003-layout}).
    \item Lay out the tabs with default options, then with a fixed width of
      900 and height of 255 twips, then with \texttt{MultiRow} and a total
      width of 1000 twips, then without \texttt{MultiRow}.
    \item Lay out the buttons with default options, with a total width of 1000
      twips (wrappable, then not wrappable) and with \texttt{ButtonWidth} 500
      and \texttt{ButtonHeight} 400.
    \item Ask for the group around several positions of a list of button
      styles.
  \end{enumerate}
}

\def\TCExpectedResult{%
  \begin{enumerate}
    \item A tab is its caption width plus 13 pixels and 18 pixels high,
      starting 2 pixels inside the control; a long caption widens the tab; a
      hidden tab is empty and takes no room.
    \item Fixed tabs are 60 by 17 pixels; \texttt{MultiRow} wraps to a new row
      at 20 pixels and no wrap happens without it.
    \item A default button is 36 by 38 pixels, a separator and a placeholder
      are at least 8 pixels wide, a button with its own width keeps it, a
      hidden button is empty; a full toolbar wraps only while
      \texttt{Wrappable} is set; fixed sizes apply.
    \item The group of a button is the run of group buttons around it, a
      button before a group joins it, a lone button is a group of one.
    \item The program prints \texttt{layout: ok} and exits with code 0.
  \end{enumerate}
}

\def\TCPostConditions{%
  The process ends; nothing is written outside CTest's own log.
}
