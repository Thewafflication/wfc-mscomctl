\def\TCID{%
  TC-0019
}

\def\TCTitle{%
  Toolbar rules
}

\def\TCPurpose{%
  Verify the toolbar button rules through the form: invalid styles and values,
  group and check presses, visibility, wrapping, the save and restore text,
  and the events of the scripted buttons.
}

\def\TCPriority{%
  \TCPriorityHigh
}

\def\TCRequirementRef{%
  REQ-0012, REQ-0020, REQ-0022
}

\def\TCDesignRef{%
  \texttt{ADR-0006} and \texttt{ADR-0010} in WFC
}

\def\TCFeatureRef{%
  \texttt{tests/corpus/223-comctl-toolbar-rules.frm} and
  \texttt{tests/corpus/223-comctl-toolbar-rules.vbp}; the form code and the
  controls of \texttt{include/wfc/mscomctl/mscomctl.inc} and the runner
  \texttt{tests/tc-0019-comctl-toolbar-rules.ps1}
}

\def\TCTestTechnique{%
  Scenario-based testing of one form: each Immediate-window line is compared
  with the recorded outcome (ISO/IEC/IEEE 29119-4 experience-based and
  specification-based techniques, with expected output comparison).
}

\def\TCPreconditions{%
  A \texttt{wfc} of the same version as the library (WFC, the interpreter and
  compiler these controls belong to, runs the case: the library holds the
  program, WFC the runner). The files of the case are in
  \texttt{tests/corpus}: \texttt{223-comctl-toolbar-rules.vbp},
  \texttt{223-comctl-toolbar-rules.frm},
  \texttt{223-comctl-toolbar-rules.expected},
  \texttt{223-comctl-toolbar-rules.debug},
  \texttt{223-comctl-toolbar-rules.ui}.
}

\def\TCEnvironment{%
  Windows with \texttt{wfc.exe} of the x64 Debug build; the form runs in WFC's
  headless host (no desktop is needed). WFC registers the case with CTest and
  runs \texttt{wfc --interpret 223-comctl-toolbar-rules.vbp}.
}

\def\TCAssumptions{%
  The scripted input \texttt{223-comctl-toolbar-rules.ui} is the only input;
  the case does not read the registry or other files except the picture named
  in the form.
}

\def\TCInputData{%
  A form with a Toolbar holding off, separator, placeholder, check, two
  groups, plain and drop-down buttons; the script presses each button, opens
  the menus and presses index 2 and 99.
}

\def\TCInitialState{%
  A fresh \texttt{wfc} process; the program creates its own controls from the
  form file and starts with no state from a previous run.
}

\def\TCProcedure{%
  \begin{enumerate}
    \item Print the state of every button at the start; set styles -1 and 6,
      values 2 and -1, a text style, a missing button and index 0.
    \item Press group buttons, a second group, release by value, and press a
      check button; print the state after each.
    \item Hide a button, read the separator width, toggle \texttt{Wrappable},
      save and restore the toolbar by key, restore a missing key and call
      \texttt{SaveToolbar} with too few arguments.
    \item Run the script.
  \end{enumerate}
}

\def\TCExpectedResult{%
  \begin{enumerate}
    \item Invalid styles and values are 380 \texttt{Invalid property value}, a
      text style is 13, a missing button is 35601 and index 0 is 35600.
    \item A group press releases its neighbours within the group; a check
      button toggles on its own.
    \item A hidden button has no width; the separator is 120.189 twips wide;
      the wrap checks with and without \texttt{Wrappable} print True; restore
      brings back the saved flags and a missing key changes nothing;
      \texttt{SaveToolbar} with too few arguments is 450 \texttt{Wrong number
      of arguments}.
    \item Events from the script: \texttt{click chk 0}, \texttt{click chk 1},
      \texttt{click ga 1}, \texttt{click gb 1} twice, \texttt{dropdown dd} and
      \texttt{menu m1One}; the disabled, separator and placeholder buttons,
      the indexes 2 and 99, a drop-down of a plain or disabled button and a
      disabled, unknown or misplaced menu raise nothing.
    \item \texttt{wfc} exits with code 0, writes the exact text of
      \texttt{223-comctl-toolbar-rules.expected} to standard output, and
      writes the exact Immediate-window text of
      \texttt{223-comctl-toolbar-rules.debug} to standard error. Any
      difference fails the case.
  \end{enumerate}
}

\def\TCPostConditions{%
  The form is closed by the last \texttt{close Form1} line of the script;
  nothing is left running and no file is written.
}
