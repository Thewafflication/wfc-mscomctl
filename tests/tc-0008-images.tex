\begin{document}

\def\TCID{%
  TC-0008
}

\def\TCTitle{%
  ImageList and ImageCombo rules
}

\def\TCPurpose{%
  Verify the rules of an ImageList and an ImageCombo: bound-list removal, the
  list size, Overlay arguments, text matching and the selected text of a
  combo.
}

\def\TCPriority{%
  \TCPriorityHigh
}

\def\TCRequirementRef{%
  REQ-0010, REQ-0015, REQ-0035, REQ-0036, REQ-0040
}

\def\TCDesignRef{%
  Not applicable; the rules are in the named header.
}

\def\TCFeatureRef{%
  \texttt{include/wfc/mscomctl/images.hpp} and \texttt{tests/images\_test.cpp}
  and the runner \texttt{tests/tc-0008-images.ps1}
}

\def\TCTestTechnique{%
  Equivalence partitioning and boundary value analysis of the rule functions
  (ISO/IEC/IEEE 29119-4 specification-based techniques).
}

\def\TCPreconditions{%
  The library is built standalone (\texttt{cmake -S . -B out/build}, then
  \texttt{cmake --build out/build --config Debug}) with
  \texttt{WFC\_BUILD\_TESTS} on; CTest runs the test as
  \texttt{TC-0008-images}.
}

\def\TCEnvironment{%
  Any target the library builds for (Windows x86, x64 and ARM64 with MSVC,
  Linux through Winelib); local runs use the x64 Debug configuration of the
  standalone build. The test needs no window, registry or network.
}

\def\TCAssumptions{%
  The rule header under test is the one built into the library; the test is
  compiled from the same source revision.
}

\def\TCInputData{%
  The himetric sizes 2540, 1270 and 0; the texts \texttt{Green},
  \texttt{GREEN}, \texttt{Blu}, \texttt{Blue}; the list \texttt{Red},
  \texttt{green}, \texttt{GREEN}; the text \texttt{abcdef} with several starts
  and lengths.
}

\def\TCInitialState{%
  A fresh process; the test keeps no state between runs and holds no files or
  handles.
}

\def\TCProcedure{%
  \begin{enumerate}
    \item Run \texttt{mscomctl\_images\_test} (CTest \texttt{TC-0008-images}).
    \item Ask whether images may be removed from a bound and an unbound list,
      whether the size may be set, and which image sets it.
    \item Convert himetric to pixels and check the Overlay argument count.
    \item Compare texts, find the first item with a text, take a selected text
      and replace a selection.
  \end{enumerate}
}

\def\TCExpectedResult{%
  \begin{enumerate}
    \item A bound list refuses removal with 35615 and the message
      \texttt{Images cannot be removed from a bound ImageList}; an unbound
      list allows it.
    \item The size may be set only while the list is empty and only the first
      image sets it; 2540 himetric is 96 pixels and 1270 is 48.
    \item Overlay takes exactly two images; texts match without regard to case
      and the first matching item is found; no item matches a partial or empty
      text.
    \item A selection is cut to the text (start beyond the text and negative
      values give the empty text); replacing inserts at the selection, or
      appends beyond the end.
    \item The program prints \texttt{images: ok} and exits with code 0.
  \end{enumerate}
}

\def\TCPostConditions{%
  The process ends; nothing is written outside CTest's own log.
}

\TCTemplate
