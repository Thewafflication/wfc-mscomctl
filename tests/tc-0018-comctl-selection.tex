\begin{document}

\def\TCID{%
  TC-0018
}

\def\TCTitle{%
  Tab selection, panel limit and bound list
}

\def\TCPurpose{%
  Verify which tab is selected after tabs are added, removed and cleared, the
  sixteen-panel limit and the rule that a bound ImageList only grows at the
  end.
}

\def\TCPriority{%
  \TCPriorityHigh
}

\def\TCRequirementRef{%
  REQ-0011, REQ-0017, REQ-0023, REQ-0035
}

\def\TCDesignRef{%
  \texttt{ADR-0006} and \texttt{ADR-0010} in WFC
}

\def\TCFeatureRef{%
  \texttt{tests/corpus/222-comctl-selection.frm} and
  \texttt{tests/corpus/222-comctl-selection.vbp}; the form code and the
  controls of \texttt{include/wfc/mscomctl/mscomctl.inc} and the runner
  \texttt{tests/tc-0018-comctl-selection.ps1}
}

\def\TCTestTechnique{%
  Scenario-based testing of one form: each Immediate-window line is compared
  with the recorded outcome (ISO/IEC/IEEE 29119-4 experience-based and
  specification-based techniques, with expected output comparison).
}

\def\TCPreconditions{%
  A \texttt{wfc} of the same version as the library (WFC, the interpreter and
  compiler these controls belong to, runs the case: the library holds the
  program, WFC the runner). The files of the case are in
  \texttt{tests/corpus}: \texttt{222-comctl-selection.vbp},
  \texttt{222-comctl-selection.frm}, \texttt{222-comctl-selection.expected},
  \texttt{222-comctl-selection.debug}, \texttt{222-comctl-selection.red.bmp}.
}

\def\TCEnvironment{%
  Windows with \texttt{wfc.exe} of the x64 Debug build; the form runs in WFC's
  headless host (no desktop is needed). WFC registers the case with CTest and
  runs \texttt{wfc --interpret 222-comctl-selection.vbp}.
}

\def\TCAssumptions{%
  No scripted input is used; the case reads only its own files.
}

\def\TCInputData{%
  A form with a TabStrip, a StatusBar and an ImageList; tabs \texttt{a} to
  \texttt{d}, \texttt{x}, \texttt{y}, \texttt{z}, \texttt{w}; seventeen panel
  additions; images added before and after the list is bound.
}

\def\TCInitialState{%
  A fresh \texttt{wfc} process; the program creates its own controls from the
  form file and starts with no state from a previous run.
}

\def\TCProcedure{%
  \begin{enumerate}
    \item Add tab \texttt{a} and print the selection; add three more.
    \item Remove the selected first tab, the selected middle tab, an
      unselected tab and the last tab, printing the selection.
    \item Add two tabs, clear, add one, add a second and remove the selected
      last one.
    \item Add seventeen panels and print the error and the count.
    \item Add two images, bind the list to the TabStrip, insert at index 1 and
      then append.
  \end{enumerate}
}

\def\TCExpectedResult{%
  \begin{enumerate}
    \item The first tab added is selected (\texttt{a}); the selection after
      the removals is \texttt{b}, \texttt{b}, \texttt{b} (count 1), then none;
      after \texttt{Clear} nothing is selected and the next tab added is
      selected (\texttt{z}); removing the last selected of several selects the
      previous (\texttt{z}).
    \item The seventeenth panel gives 35609 \texttt{A status bar holds at most
      sixteen panels}; sixteen panels remain.
    \item Before binding an insert in front works (key \texttt{front} first);
      after binding an insert is 35615 \texttt{Images can only be appended to
      a bound ImageList} and an append works (3 images).
    \item \texttt{wfc} exits with code 0, writes the exact text of
      \texttt{222-comctl-selection.expected} to standard output, and writes
      the exact Immediate-window text of \texttt{222-comctl-selection.debug}
      to standard error. Any difference fails the case.
  \end{enumerate}
}

\def\TCPostConditions{%
  The form is closed by the last \texttt{close Form1} line of the script;
  nothing is left running and no file is written.
}

\TCTemplate
